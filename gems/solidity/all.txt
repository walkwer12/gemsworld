Aim : Palindrome


pragma solidity ^0.8.21;

contract Palindrome {

    function check(uint n) public pure returns (string memory) {

        uint r;
        uint rev = 0;
        uint n1 = n;

        while (n != 0) {

            r = n % 10;
            rev = rev * 10 + r;
            n = n / 10;

        }

        if (rev == n1)
            return "Palindrome";
        else
            return "Not Palindrome";

    }

}

        uint id;
        string name;
        uint balance;
    }

    }

    }

 //SPDX-License-Identifier: MIT
 pragma solidity ^0.8.21;

 // Perfect Number

 contract PerfectNumber {

    function calcPerfectNum(uint num) public pure returns(bool) {

        if (num <= 1) {
            return false;
        }

        uint sum = 0;

        for (uint i = 1; i <= num / 2; i++) {
            if (num % i == 0) {
                sum += i;
            }
        }

        return sum == num;
    }
 }

contract voting {

    string[] public cn;

    constructor(string[] memory cdn) {
        cn = cdn;
    }

    mapping(string => uint) public ct;
    mapping(address => bool) public voters;

    function vote(string memory c) public {

        require(
            !voters[msg.sender],
            "you have alredy voted"
        );

        bool ce = false;

        for (uint i = 0; i < cn.length; i++) {
            if (
                keccak256(bytes(c)) ==
                keccak256(bytes(cn[i]))
            ) {
                ce = true;
                break;
            }
        }

        require(
            ce,
            "Condidate dows not exists"
        );

        ct[c]++;
        voters[msg.sender] = true;
    }

    function getWinner()
        public
        view
        returns (string memory)
    {
        string memory w;
        uint t = 0;

        for (uint i = 0; i < cn.length; i++) {

            if (ct[cn[i]] > t) {
                t = ct[cn[i]];
                w = cn[i];
            }
        }

        return w;
    }
}

Aim: Student Structure and grade management


// SPDX-License-Identifier: MIT
pragma solidity ^0.8.21;

contract StudentGrade {

    // Student structure
    struct Student {
        uint id;
        string name;
        string s_address;
        uint age;
        uint[] marks;
    }

    // Mapping student ID => Student
    mapping(uint => Student) public stud_id;

    // Store student IDs
    uint[] public studentIds;

    // Total number of students
    uint public studentCount;

    // Add a student
    function addStudent(
        uint id,
        string memory name,
        string memory s_address,
        uint age,
        uint[] calldata marks
    ) public {

        // Prevent duplicate student IDs
        require(
            stud_id[id].id == 0,
            "Student ID already exists"
        );

        // Store student details
        stud_id[id] = Student(
            id,
            name,
            s_address,
            age,
            marks
        );

        // Store ID separately
        studentIds.push(id);

        // Increase student count
        studentCount++;
    }

    // Total number of students
    function totalStudent()
        public
        view
        returns (uint)
    {
        return studentCount;
    }

    // Display details of all students
    function getAllStudents()
        public
        view
        returns (Student[] memory)
    {
        Student[] memory allStudents =
            new Student[](studentCount);

        for (uint i = 0; i < studentCount; i++) {

            uint id = studentIds[i];

            allStudents[i] = stud_id[id];
        }

        return allStudents;
    }

    // Calculate grade of a student
    function getGrade(uint id)
        public
        view
        returns (string memory)
    {
        // Check that student exists
        require(
            stud_id[id].id != 0,
            "Student does not exist"
        );

        uint[] memory marks = stud_id[id].marks;

        // Prevent division by zero
        if (marks.length == 0) {
            return "No marks";
        }

        uint totalMarks = 0;

        // Calculate total marks
        for (uint i = 0; i < marks.length; i++) {
            totalMarks += marks[i];
        }

        // Calculate average
        uint averageMark =
            totalMarks / marks.length;

        // Grade calculation
        if (averageMark >= 90) {
            return "Grade: A+";
        }
        else if (averageMark >= 80) {
            return "Grade: A";
        }
        else if (averageMark >= 70) {
            return "Grade: B";
        }
        else if (averageMark >= 60) {
            return "Grade: C";
        }
        else if (averageMark >= 50) {
            return "Grade: D";
        }
        else {
            return "Grade: F";
        }
    }

    // Display details of a particular student
    function displayDetail(uint id)
        public
        view
        returns (
            uint,
            string memory,
            string memory,
            uint,
            uint[] memory
        )
    {
        // Check that student exists
        require(
            stud_id[id].id != 0,
            "Student does not exist"
        );

        Student memory s = stud_id[id];

        return (
            s.id,
            s.name,
            s.s_address,
            s.age,
            s.marks
        );
    }
}

        string Qtest;
        string CA;
    }

    uint public tc;
    uint public tw;

    event AnswerE(string qu, string uans, bool isc);

    function add(
        string memory qt,
        string memory cans
    ) public {
        q.push(Question(qt, cans));
    }

    function ansQ(
        uint i,
        string memory uans
    ) public {

        require(i < q.length, "Invalid Question Index");

        if (
            keccak256(bytes(uans)) ==
            keccak256(bytes(q[i].CA))
        ) {
            tc++;
            emit AnswerE(q[i].Qtest, uans, true);
        }
        else {
            tw++;
            emit AnswerE(q[i].Qtest, uans, false);
        }
    }

aim : Vehicle Rental System with Function Overriding


// SPDX-License-Identifier: MIT
pragma solidity ^0.8.21;

contract Rental {
    string public agencyName;

    event VehicleRented(
        string vehicleType,
        uint daysRented,
        uint totalRent
    );

    constructor(string memory name) {
        agencyName = name;
    }

    function calculateRent(uint daysRented)
        public
        view
        virtual
        returns (uint)
    {
        return daysRented * 1000;
    }
}

contract CarRental is Rental {

    constructor(string memory name) Rental(name) {}

    function calculateRent(uint daysRented)
        public
        view
        override
        returns (uint)
    {
        return daysRented * 2000;
    }

    function rentVehicle(uint daysRented) public {
        uint rent = calculateRent(daysRented);
        emit VehicleRented("Car", daysRented, rent);
    }
}

contract BikeRental is Rental {

    constructor(string memory name) Rental(name) {}

    function calculateRent(uint daysRented)
        public
        view
        override
        returns (uint)
    {
        return daysRented * 500;
    }

    function rentVehicle(uint daysRented) public {
        uint rent = calculateRent(daysRented);
        emit VehicleRented("Bike", daysRented, rent);
    }
}
















































































































aim : crowdfunding contracts


// SPDX-License-Identifier: MIT
pragma solidity ^0.8.21;

contract Crowdfunding {
    address public owner;
    uint public fundingGoal;
    uint public deadline;
    uint public totalFunds;

    event ContributionReceived(address, uint);
    event GoalReached(uint);

    constructor(uint _fundingGoal, uint _duration) {
        owner = msg.sender;
        fundingGoal = _fundingGoal;
        deadline = block.timestamp + _duration;
    }

    function dispo() public view returns (address) {
        return owner;
    }
}

contract CrowdfundingCampaign is Crowdfunding {

    constructor(uint _fundingGoal, uint _duration)
        Crowdfunding(_fundingGoal, _duration) {}

    function contribute() public payable {
        require(block.timestamp < deadline, "Campaign ended");
        require(msg.value > 0, "Invalid contribution");

        totalFunds += msg.value;

        emit ContributionReceived(msg.sender, msg.value);

        if (totalFunds >= fundingGoal) {
            emit GoalReached(totalFunds);
        }
    }

    function refund() public payable {
        require(block.timestamp >= deadline, "Campaign not ended");
        require(totalFunds < fundingGoal, "Goal reached");

        uint amount = address(this).balance;
        payable(msg.sender).transfer(amount);
    }
}
